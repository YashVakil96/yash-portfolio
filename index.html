❌ Legacy MonoBehaviour (28.4ms) 42KB GC/f 
public class SwarmEnemy_Legacy : MonoBehaviour
{
    void Update()
    {
        // ❌ Slow scene search + LINQ heap allocation
        var player = FindObjectOfType<Player>();
        var hits = Physics.OverlapSphere(transform.position, 8f)
                          .ToList(); // ❌ GC Spike!

        // ❌ Unbatched main-thread transform marshaling
        Vector3 dir = (player.transform.position - transform.position).normalized;
        transform.position += dir * (6f * Time.deltaTime);
    }
}
✅ Yash's Burst + Jobs Standard (0.4ms) 0 Bytes GC 
[BurstCompile(FloatMode = FloatMode.Fast)]
public struct SwarmSteerJob : IJobParallelForTransform
{
    [ReadOnly] public float3 TargetPos;
    [ReadOnly] public float DeltaTime;

    public void Execute(int index, TransformAccess tx)
    {
        // ✅ Multi-threaded SIMD math & zero GC allocation
        float3 pos = tx.position;
        float3 dir = math.normalizesafe(TargetPos - pos);
        tx.position = pos + dir * (6f * DeltaTime);
    }
}
